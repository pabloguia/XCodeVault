# Script: scripts/experiments/e12-case-sensitivity.sh @ 47b0462
# Runner: <user>; sudo: not used — hdiutil needs no privileges for a sparse image

## internal volume is case-insensitive (the baseline this is measured against)
$ sh -c diskutil info / | grep -i 'File System Personality' | sed 's/^ *//'
File System Personality:   APFS
[exit=0]

## setup — disposable case-sensitive APFS image
## create
$ hdiutil create -size 6g -fs Case-sensitive APFS -volname XCVE12 -type SPARSE /private/tmp/xcv-e12
created: /private/tmp/xcv-e12.sparseimage
[exit=0]

## attach at a private mount point
$ hdiutil attach /private/tmp/xcv-e12.sparseimage -nobrowse -mountpoint /private/tmp/xcv-e12-mnt
/dev/disk8          	GUID_partition_scheme          	
/dev/disk8s1        	EFI                            	
/dev/disk8s2        	Apple_APFS                     	
/dev/disk9          	EF57347C-0000-11AA-AA11-0030654	
/dev/disk9s1        	41504653-0000-11AA-AA11-0030654	/private/tmp/xcv-e12-mnt
[exit=0]

## confirm we are on the image we just made, not somebody else's volume
$ sh -c diskutil info '/private/tmp/xcv-e12-mnt' | grep -i 'Volume Name' | sed 's/^ *//'
Volume Name:               XCVE12
[exit=0]

## what we got
$ sh -c diskutil info /private/tmp/xcv-e12-mnt | grep -Ei 'File System Personality|Owners|Mount Point' | sed 's/^ *//'

File System Personality:   Case-sensitive APFS
Owners:                    Disabled
[exit=0]

## control — prove the two volumes really differ, rather than assuming it
## two names differing only in case, on the case-sensitive volume
$ sh -c mkdir -p /private/tmp/xcv-e12-mnt/casecheck && printf 'lower\n' > /private/tmp/xcv-e12-mnt/casecheck/probe.txt && printf 'upper\n' > /private/tmp/xcv-e12-mnt/casecheck/PROBE.txt && ls /private/tmp/xcv-e12-mnt/casecheck/
PROBE.txt
probe.txt
[exit=0]

## the same two names on the internal case-insensitive volume
$ sh -c d=$(mktemp -d); printf 'lower\n' > $d/probe.txt; printf 'upper\n' > $d/PROBE.txt; ls $d; echo '--- surviving content ---'; cat $d/probe.txt; rm -rf $d
probe.txt
--- surviving content ---
upper
[exit=0]
